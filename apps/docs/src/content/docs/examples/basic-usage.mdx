## Example 1: Creating Component Specifications

### Simple Button Component

Let's start with a basic button component specification:

```typescript
import { parseUISpec } from '@reallygoodwork/coral-core'

// Define a simple button specification
const buttonSpec = {
  elementType: 'button',
  name: 'SimpleButton',
  children: ['Click me!'],
  styles: {
    backgroundColor: '#3b82f6',
    color: '#ffffff',
    padding: '8px 16px',
    borderRadius: '4px',
    border: 'none',
    fontSize: '14px',
    fontWeight: '500',
    cursor: 'pointer',
    transition: 'background-color 0.2s ease'
  },
  attributes: {
    type: 'button'
  }
}

// Validate the specification
try {
  const validatedSpec = parseUISpec(buttonSpec)
  console.log('✅ Button spec is valid!')
  console.log('Component name:', validatedSpec.name)
  console.log('Element type:', validatedSpec.elementType)
} catch (error) {
  console.error('❌ Validation failed:', error.message)
}
```

## Example 3: Tailwind CSS Conversion

### Basic Style Conversion

Convert Tailwind classes to CSS properties:

```typescript
import { tailwindToCSS } from '@reallygoodwork/tw2css'

// Convert common utility classes
const examples = [
  'bg-blue-500 text-white px-4 py-2 rounded-md',
  'flex items-center justify-between w-full',
  'grid grid-cols-1 md:grid-cols-2 lg:grid-cols-3 gap-6',
  'text-xl font-bold text-gray-800 mb-4',
  'border border-gray-300 rounded-lg p-6 shadow-sm'

examples.forEach((classes, index) => {
  const cssStyles = tailwindToCSS(classes)
  console.log(`\nExample ${index + 1}: "${classes}"`)
  console.log('CSS Output:', JSON.stringify(cssStyles, null, 2))
})
```

## Example 5: Real-World Integration

### Build Tool Integration

Here's how you might integrate Coral UI into a build process:

```typescript
// build-tools/coral-processor.ts
import { transformHTMLToSpec, parseUISpec } from '@reallygoodwork/coral-core'
import { tailwindToCSS } from '@reallygoodwork/tw2css'
import fs from 'fs/promises'
import path from 'path'

interface ProcessingOptions {
  inputDir: string
  outputDir: string
  convertTailwind: boolean
}

export async function processComponentFiles(options: ProcessingOptions) {
  const { inputDir, outputDir, convertTailwind } = options
  
  try {
    // Read all HTML files from input directory
    const files = await fs.readdir(inputDir)
    const htmlFiles = files.filter(file => file.endsWith('.html'))
    
    console.log(`Processing ${htmlFiles.length} HTML files...`)
    
    const results = []
    
    for (const file of htmlFiles) {
      const filePath = path.join(inputDir, file)
      const htmlContent = await fs.readFile(filePath, 'utf-8')
      
      try {
        // Transform HTML to Coral spec
        let spec = transformHTMLToSpec(htmlContent)
        
        // Convert Tailwind classes if requested
        if (convertTailwind && spec.attributes?.class) {
          const tailwindClasses = spec.attributes.class
          const cssStyles = tailwindToCSS(tailwindClasses)
          
          // Merge converted styles
          spec.styles = { ...spec.styles, ...cssStyles }
          
          // Remove class attribute
          delete spec.attributes.class
        }
        
        // Validate the final spec
        const validatedSpec = parseUISpec(spec)
        
        // Write to output directory
        const outputFile = file.replace('.html', '.coral.json')
        const outputPath = path.join(outputDir, outputFile)
        
        await fs.writeFile(
          outputPath, 
          JSON.stringify(validatedSpec, null, 2)
        )
        
        results.push({
          file,
          success: true,
          outputFile
        })
        
        console.log(`✅ Processed: ${file} → ${outputFile}`)
        
      } catch (error) {
        console.error(`❌ Failed to process ${file}:`, error.message)
        results.push({
          file,
          success: false,
          error: error.message
        })
      }
    }
    
    return results
    
  } catch (error) {
    console.error('Processing failed:', error)
    throw error
  }
}

// Usage
processComponentFiles({
  inputDir: './src/components',
  outputDir: './dist/specs',
  convertTailwind: true
}).then(results => {
  const successful = results.filter(r => r.success).length
  console.log(`\n🎉 Successfully processed ${successful}/${results.length} files`)
})
```

## Best Practices

<Aside type="tip" title="Development Tips">
1. **Always validate specs** with `parseUISpec()` during development
2. **Use TypeScript** for better development experience and type safety
3. **Start simple** and gradually add complexity to your specifications
4. **Cache results** when processing multiple files or specs
5. **Handle errors gracefully** with proper try-catch blocks
</Aside>

<Aside type="note" title="Performance Notes">
- Coral UI is optimized for build-time processing
- Consider using workers for CPU-intensive transformations
- Batch similar operations when possible
- Use incremental processing for large component libraries
</Aside>
