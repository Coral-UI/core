Coral UI uses comprehensive JSON schemas to ensure specification correctness, provide type safety, and enable robust validation across the entire ecosystem.

## Overview

The schema system provides:

- **Validation** - Ensure specifications are well-formed and complete
- **Type Safety** - Generate TypeScript types from schemas
- **Documentation** - Self-documenting specification format
- **IDE Support** - Autocomplete and error detection
- **Interoperability** - Standard JSON Schema format

| `ts-types.schema.json` | TypeScript types | Type definitions, generics |

## Using Schemas

### Validation with Core Functions

The primary way to validate specifications is through the core validation functions:

```typescript
import { parseUISpec } from '@reallygoodwork/coral-core'

const spec = {
  elementType: 'button',
  name: 'MyButton',
  children: ['Click me']
}

try {
  const validatedSpec = parseUISpec(spec)
  console.log('✅ Specification is valid')
} catch (error) {
  console.error('❌ Validation failed:', error.message)
}
```

## Custom Validation

### Extending Schemas

Create custom validation rules:

```typescript
import Ajv from 'ajv'
import { exportSchemaToJSON } from '@reallygoodwork/coral-core'

function createCustomValidator() {
  const ajv = new Ajv()
  
  // Add custom format for CSS values
  ajv.addFormat('css-length', {
    type: 'string',
    validate: (value: string) => {
      return /^\d+(px|em|rem|%|vh|vw)$/.test(value)
    }
  })
  
  // Add custom keyword for design system tokens
  ajv.addKeyword({
    keyword: 'designToken',
    type: 'string',
    compile: (schema: any) => {
      return (value: string) => {
        return value.startsWith('$') || value.startsWith('--')
      }
    }
  })
  
  const baseSchema = exportSchemaToJSON('coral')
  
  // Extend schema with custom rules
  const customSchema = {
    ...baseSchema,
    properties: {
      ...baseSchema.properties,
      styles: {
        ...baseSchema.properties.styles,
        properties: {
          ...baseSchema.properties.styles.properties,
          // Add custom validation for specific CSS properties
          fontSize: {
            type: 'string',
            format: 'css-length'
          },
          color: {
            type: 'string',
            designToken: true
          }
        }
      }
    }
  }
  
  return ajv.compile(customSchema)
}

const customValidate = createCustomValidator()

const spec = {
  elementType: 'p',
  name: 'Text',
  styles: {
    fontSize: '16px',    // ✅ Valid CSS length
    color: '$primary'    // ✅ Valid design token
  }
}

if (customValidate(spec)) {
  console.log('✅ Custom validation passed')
} else {
  console.error('❌ Custom validation failed:', customValidate.errors)
}
```

### Validation Middleware

Create validation middleware for different contexts:

```typescript
interface ValidationContext {
  framework?: 'react' | 'vue' | 'angular'
  target?: 'web' | 'mobile' | 'email'
  strict?: boolean
}

function createContextualValidator(context: ValidationContext) {
  const baseSchema = exportSchemaToJSON('coral')
  
  // Modify schema based on context
  let contextualSchema = { ...baseSchema }
  
  if (context.framework === 'react') {
    // Add React-specific validations
    contextualSchema.properties.stateHooks = {
      type: 'array',
      items: { $ref: '#/definitions/stateHook' }
    }
  }
  
  if (context.target === 'email') {
    // Restrict to email-safe CSS properties
    contextualSchema.properties.styles.properties = {
      // Only allow email-safe properties
      backgroundColor: { type: 'string' },
      color: { type: 'string' },
      fontSize: { type: 'string' },
      fontFamily: { type: 'string' },
      padding: { type: 'string' },
      margin: { type: 'string' }
    }
  }
  
  if (context.strict) {
    // Enable strict mode validations
    contextualSchema.additionalProperties = false
    contextualSchema.required = [...contextualSchema.required, 'description']
  }
  
  const ajv = new Ajv()
  return ajv.compile(contextualSchema)
}

// Usage examples
const reactValidator = createContextualValidator({ 
  framework: 'react', 
  target: 'web' 
})

const emailValidator = createContextualValidator({ 
  target: 'email', 
  strict: true 
})
```

## Error Handling Best Practices

### Structured Error Reporting

```typescript
interface ValidationResult {
  valid: boolean
  errors?: ValidationError[]
  warnings?: ValidationWarning[]
}

function validateWithDetails(spec: unknown): ValidationResult {
  try {
    const validated = parseUISpec(spec)
    return { valid: true }
  } catch (error) {
    if (error.name === 'ValidationError') {
      return {
        valid: false,
        errors: error.errors.map(err => ({
          path: err.path,
          message: err.message,
          value: err.value,
          severity: 'error'
        }))
      }
    }
    
    return {
      valid: false,
      errors: [{
        path: 'root',
        message: error.message,
        severity: 'error'
      }]
    }
  }
}
```

### Progressive Validation

```typescript
function progressiveValidation(spec: unknown) {
  const results = []
  
  // Step 1: Basic structure validation
  try {
    validateBasicStructure(spec)
    results.push({ stage: 'structure', status: 'passed' })
  } catch (error) {
    results.push({ stage: 'structure', status: 'failed', error })
    return results // Stop on critical errors
  }
  
  // Step 2: Property validation
  try {
    validateProperties(spec)
    results.push({ stage: 'properties', status: 'passed' })
  } catch (error) {
    results.push({ stage: 'properties', status: 'failed', error })
  }
  
  // Step 3: Semantic validation
  try {
    validateSemantics(spec)
    results.push({ stage: 'semantics', status: 'passed' })
  } catch (error) {
    results.push({ stage: 'semantics', status: 'failed', error })
  }
  
  return results
}
```

## Performance Considerations

### Schema Caching

```typescript
class SchemaCache {
  private static cache = new Map<string, any>()
  
  static getSchema(name: string) {
    if (!this.cache.has(name)) {
      const schema = exportSchemaToJSON(name)
      this.cache.set(name, schema)
    }
    return this.cache.get(name)
  }
  
  static clearCache() {
    this.cache.clear()
  }
}
```

### Validation Optimization

```typescript
// Pre-compile validators for better performance
const validators = {
  coral: (() => {
    const ajv = new Ajv()
    return ajv.compile(exportSchemaToJSON('coral'))
  })(),
  
  styles: (() => {
    const ajv = new Ajv()
    return ajv.compile(exportSchemaToJSON('styles'))
  })(),
  
  state: (() => {
    const ajv = new Ajv()
    return ajv.compile(exportSchemaToJSON('state'))
  })()
}

function fastValidate(spec: unknown, schemaName: string) {
  const validator = validators[schemaName]
  if (!validator) {
    throw new Error(`Unknown schema: ${schemaName}`)
  }
  
  return validator(spec)
}
```

## Best Practices

<Aside type="tip" title="Validation Strategy">
- **Validate early** - Check specifications as soon as they're created
- **Use progressive validation** - Check structure before detailed properties  
- **Cache validators** - Pre-compile schemas for better performance
- **Provide clear errors** - Help developers understand what went wrong
</Aside>

<Aside type="note" title="Schema Design">
- **Keep schemas focused** - Each schema should have a single responsibility
- **Use references** - Avoid duplicating schema definitions
- **Document thoroughly** - Include descriptions for all properties
- **Version schemas** - Plan for backward compatibility
</Aside>
